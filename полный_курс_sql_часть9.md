# Раздел 10: Мастерство SQL

## Глава 76: Профессиональные паттерны SQL

### Введение

Профессиональные паттерны SQL - это проверенные на практике подходы и шаблоны, которые позволяют писать более эффективный, читаемый и поддерживаемый SQL-код. Знание этих паттернов помогает решать сложные задачи быстрее и качественнее, а также позволяет избежать распространенных ошибок.

### Паттерн 1: Использование CTE для структурирования сложных запросов

#### Проблема
Сложные запросы с множественными подзапросами становятся трудночитаемыми и сложными для отладки.

#### Решение
Разбиение сложного запроса на логические блоки с помощью CTE.

```sql
-- Неправильно: вложенные подзапросы без структуры
SELECT 
    фамилия,
    имя,
    (SELECT SUM(общая_сумма) FROM заказы WHERE id_клиента = клиенты.id) AS общая_сумма,
    (SELECT COUNT(*) FROM заказы WHERE id_клиента = клиенты.id) AS количество_заказов
FROM клиенты
WHERE (SELECT COUNT(*) FROM заказы WHERE id_клиента = клиенты.id) > 5;

-- Правильно: структурированный запрос с CTE
WITH клиентская_статистика AS (
    SELECT 
        id_клиента,
        COUNT(*) AS количество_заказов,
        SUM(общая_сумма) AS общая_сумма
    FROM заказы
    GROUP BY id_клиента
),
отобранные_клиенты AS (
    SELECT *
    FROM клиентская_статистика
    WHERE количество_заказов > 5
)
SELECT 
    к.фамилия,
    к.имя,
    oc.общая_сумма,
    oc.количество_заказов
FROM клиенты к
JOIN отобранные_клиенты oc ON к.id = oc.id_клиента
ORDER BY oc.общая_сумма DESC;
```

#### Продвинутый пример: многоуровневый CTE

```sql
WITH 
-- Этап 1: Подготовка основных данных
базовые_данные AS (
    SELECT 
        к.id AS id_клиента,
        к.фамилия,
        к.имя,
        к.город,
        з.id AS id_заказа,
        з.дата_заказа,
        з.общая_сумма
    FROM клиенты к
    LEFT JOIN заказы з ON к.id = з.id_клиента
    WHERE з.дата_заказа >= CURRENT_DATE - INTERVAL '1 year'
),

-- Этап 2: Агрегация по клиентам
клиентская_статистика AS (
    SELECT 
        id_клиента,
        COUNT(дата_заказа) AS количество_заказов,
        SUM(общая_сумма) AS общая_сумма,
        AVG(общая_сумма) AS средний_чек,
        MIN(дата_заказа) AS первый_заказ,
        MAX(дата_заказа) AS последний_заказ
    FROM базовые_данные
    GROUP BY id_клиента
),

-- Этап 3: Классификация клиентов
категоризованные_клиенты AS (
    SELECT 
        *,
        CASE 
            WHEN общая_сумма > 200000 THEN 'VIP'
            WHEN общая_сумма > 100000 THEN 'Премиум'
            WHEN количество_заказов > 10 THEN 'Частый клиент'
            ELSE 'Обычный'
        END AS категория_клиента
    FROM клиентская_статистика
),

-- Этап 4: Географический анализ
городская_статистика AS (
    SELECT 
        к.город,
        AVG(cs.общая_сумма) AS средняя_выручка_на_клиента,
        COUNT(*) AS количество_клиентов,
        SUM(cs.общая_сумма) AS общая_выручка
    FROM клиенты к
    JOIN клиентская_статистика cs ON к.id = cs.id_клиента
    GROUP BY к.город
)

-- Финальный результат
SELECT 
    к.фамилия,
    к.имя,
    к.город,
    ck.количество_заказов,
    ck.общая_сумма,
    ck.категория_клиента,
    gs.средняя_выручка_на_клиента AS средняя_выручка_в_городе
FROM клиенты к
JOIN категоризованные_клиенты ck ON к.id = ck.id_клиента
JOIN городская_статистика gs ON к.город = gs.город
ORDER BY ck.общая_сумма DESC
LIMIT 100;
```

### Паттерн 2: Результат-ориентированный подход с CASE

#### Проблема
Сложные условия с множественными проверками и разными результатами.

#### Решение
Использование структурированных CASE выражений.

```sql
-- Пример: Комплексная сегментация клиентов
CREATE VIEW сегментация_клиентов_с_обратной_связью AS
WITH клиентская_матрица AS (
    SELECT 
        к.id AS id_клиента,
        к.фамилия,
        к.имя,
        -- RFM-показатели
        EXTRACT(DAY FROM (CURRENT_DATE - MAX(з.дата_заказа))) AS дни_с_последнего_заказа,
        COUNT(з.id) AS количество_заказов,
        SUM(з.общая_сумма) AS общая_сумма,
        AVG(з.общая_сумма) AS средний_чек
    FROM клиенты к
    LEFT JOIN заказы з ON к.id = з.id_клиента
    WHERE к.дата_регистрации <= CURRENT_DATE - INTERVAL '30 days'  -- только старые клиенты
    GROUP BY к.id, к.фамилия, к.имя
),
сегментация AS (
    SELECT 
        *,
        -- R (Recency) - 0-2 недели: 5, 2-4 недели: 4, 1-2 месяца: 3, 2-4 месяца: 2, >4 месяцев: 1
        CASE 
            WHEN дни_с_последнего_заказа <= 14 THEN 5
            WHEN дни_с_последнего_заказа <= 28 THEN 4
            WHEN дни_с_последнего_заказа <= 60 THEN 3
            WHEN дни_с_последнего_заказа <= 120 THEN 2
            ELSE 1
        END AS оценка_давности,
        -- F (Frequency) - по количеству заказов
        CASE 
            WHEN количество_заказов >= 10 THEN 5
            WHEN количество_заказов >= 5 THEN 4
            WHEN количество_заказов >= 3 THEN 3
            WHEN количество_заказов >= 1 THEN 2
            ELSE 1
        END AS оценка_частоты,
        -- M (Monetary) - по общей сумме
        CASE 
            WHEN общая_сумма >= 500000 THEN 5
            WHEN общая_сумма >= 200000 THEN 4
            WHEN общая_сумма >= 100000 THEN 3
            WHEN общая_сумма >= 50000 THEN 2
            ELSE 1
        END AS оценка_денег
    FROM клиентская_матрица
)
SELECT 
    фамилия,
    имя,
    дни_с_последнего_заказа,
    количество_заказов,
    общая_сумма,
    оценка_давности,
    оценка_частоты,
    оценка_денег,
    -- Комбинированный RFM-сегмент
    CASE 
        WHEN оценка_давности >= 4 AND оценка_частоты >= 4 AND оценка_денег >= 4 THEN 'Чемпионы'
        WHEN оценка_давности >= 3 AND оценка_частоты >= 3 THEN 'Лояльные покупатели'
        WHEN оценка_денег >= 4 AND оценка_частоты <= 3 THEN 'Потенциальные чемпионы'
        WHEN оценка_давности >= 4 AND оценка_денег >= 3 THEN 'Новые фавориты'
        WHEN оценка_давности <= 2 THEN 'Ушедшие клиенты'
        WHEN оценка_частоты <= 2 AND оценка_денег <= 2 THEN 'Потерянные клиенты'
        WHEN оценка_давности >= 3 AND оценка_частоты <= 2 THEN 'Перспективные'
        WHEN оценка_давности <= 2 AND оценка_частоты >= 3 THEN 'Недавно ушедшие'
        ELSE 'Другие'
    END AS RFM_сегмент,
    -- Рекомендации по сегменту
    CASE 
        WHEN оценка_давности >= 4 AND оценка_частоты >= 4 AND оценка_денег >= 4 THEN 'Поддерживать привилегии, предлагать эксклюзив'
        WHEN оценка_давности >= 3 AND оценка_частоты >= 3 THEN 'Программы лояльности, персональные предложения'
        WHEN оценка_денег >= 4 AND оценка_частоты <= 3 THEN 'Частые напоминания, специальные скидки'
        WHEN оценка_давности >= 4 AND оценка_денег >= 3 THEN 'Рекомендательные программы'
        WHEN оценка_давности <= 2 THEN 'Ретаргетинг, восстановительные предложения'
        ELSE 'Обычные маркетинговые кампании'
    END AS рекомендуемая_стратегия
FROM сегментация
ORDER BY общая_сумма DESC;
```

### Паттерн 3: Логическое разбиение сложных условий

#### Проблема
Сложные условия WHERE становятся трудночитаемыми и сложноотлаживаемыми.

#### Решение
Использование CTE для разделения логики фильтрации.

```sql
-- Пример: сложная фильтрация заказов
WITH 
-- Основной фильтр по дате и статусу
базовые_условия AS (
    SELECT *
    FROM заказы
    WHERE дата_заказа >= CURRENT_DATE - INTERVAL '6 months'
      AND статус NOT IN ('отменен', 'вернут')
),

-- Фильтр по сумме заказа
фильр_по_сумме AS (
    SELECT *
    FROM базовые_условия
    WHERE общая_сумма > 10000
),

-- Фильтр по региону клиента
фильр_по_региону AS (
    SELECT фс.*
    FROM фильтр_по_сумме фс
    JOIN клиенты к ON фс.id_клиента = к.id
    WHERE к.город IN ('Москва', 'Санкт-Петербург', 'Новосибирск')
)

-- Окончательный результат
SELECT 
    fz.*,
    к.фамилия,
    к.имя
FROM фильтр_по_региону fz
JOIN клиенты к ON fz.id_клиента = к.id
ORDER BY fz.дата_заказа DESC;
```

### Паттерн 4: Матричные вычисления

#### Проблема
Необходимость сравнения данных между разными периодами или категориями.

#### Решение
Использование CASE и GROUP BY для создания матриц.

```sql
-- Пример: сводная матрица продаж по городам и месяцам
SELECT 
    к.город,
    SUM(CASE WHEN EXTRACT(MONTH FROM з.дата_заказа) = 1 THEN з.общая_сумма ELSE 0 END) AS "Январь",
    SUM(CASE WHEN EXTRACT(MONTH FROM з.дата_заказа) = 2 THEN з.общая_сумма ELSE 0 END) AS "Февраль",
    SUM(CASE WHEN EXTRACT(MONTH FROM з.дата_заказа) = 3 THEN з.общая_сумма ELSE 0 END) AS "Март",
    SUM(CASE WHEN EXTRACT(MONTH FROM з.дата_заказа) = 4 THEN з.общая_сумма ELSE 0 END) AS "Апрель",
    SUM(CASE WHEN EXTRACT(MONTH FROM з.дата_заказа) = 5 THEN з.общая_сумма ELSE 0 END) AS "Май",
    SUM(CASE WHEN EXTRACT(MONTH FROM з.дата_заказа) = 6 THEN з.общая_сумма ELSE 0 END) AS "Июнь",
    SUM(з.общая_сумма) AS "Итого"
FROM заказы з
JOIN клиенты к ON з.id_клиента = к.id
WHERE з.дата_заказа >= '2023-01-01' AND з.дата_заказа < '2023-07-01'
GROUP BY к.город
ORDER BY "Итого" DESC;
```

### Паттерн 5: Иерархические данные

#### Проблема
Работа с иерархическими данными (например, категории товаров).

#### Решение
Использование рекурсивных CTE.

```sql
-- Предположим, у нас есть иерархия категорий
CREATE TABLE категории_иерархия (
    id INTEGER PRIMARY KEY,
    название VARCHAR(100),
    родительская_категория INTEGER REFERENCES категории_иерархия(id)
);

INSERT INTO категории_иерархия VALUES
(1, 'Электроника', NULL),
(2, 'Телефоны', 1),
(3, 'Смартфоны', 2),
(4, 'Телефоны Android', 3),
(5, 'Телефоны iPhone', 3),
(6, 'Компьютеры', 1),
(7, 'Ноутбуки', 6),
(8, 'Настольные ПК', 6);

-- Рекурсивный CTE для получения полного пути категории
WITH RECURSIVE категориальная_иерархия AS (
    -- Базовый случай: корневые категории
    SELECT 
        id,
        название,
        родительская_категория,
        1 AS уровень_глубины,
        название AS полный_путь
    FROM категории_иерархия
    WHERE родительская_категория IS NULL
    
    UNION ALL
    
    -- Рекурсивный случай: дочерние категории
    SELECT 
        c.id,
        c.название,
        c.родительская_категория,
        ci.уровень_глубины + 1 AS уровень_глубины,
        ci.полный_путь || ' > ' || c.название AS полный_путь
    FROM категории_иерархия c
    JOIN категориальная_иерархия ci ON c.родительская_категория = ci.id
)
SELECT 
    уровень_глубины,
    полный_путь,
    название
FROM категориальная_иерархия
ORDER BY полный_путь;
```

### Паттерн 6: Построение скользящих окон

#### Проблема
Необходимость вычисления скользящих статистических показателей.

#### Решение
Использование оконных функций с определенными фреймами.

```sql
-- Пример: 7-дневное скользящее среднее по продажам
WITH ежедневные_продажи AS (
    SELECT 
        дата_заказа::DATE AS дата,
        SUM(общая_сумма) AS дневная_выручка
    FROM заказы
    WHERE дата_заказа >= CURRENT_DATE - INTERVAL '30 days'
    GROUP BY дата_заказа::DATE
)
SELECT 
    дата,
    дневная_выручка,
    AVG(дневная_выручка) OVER (
        ORDER BY дата
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ) AS скользящее_среднее_7_дней,
    AVG(дневная_выручка) OVER (
        ORDER BY дата
        ROWS BETWEEN 13 PRECEDING AND CURRENT ROW
    ) AS скользящее_среднее_14_дней,
    -- Отклонение от среднего
    дневная_выручка - AVG(дневная_выручка) OVER (
        ORDER BY дата
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ) AS отклонение_от_среднего
FROM ежедневные_продажи
ORDER BY дата;
```

### Паттерн 7: Условные агрегации

#### Проблема
Необходимость вычисления разных метрик на основе условий.

#### Решение
Использование CASE внутри агрегатных функций.

```sql
-- Пример: детализированный отчет по клиентам с разными метриками
SELECT 
    к.город,
    COUNT(*) AS "Всего клиентов",
    -- Условные метрики
    COUNT(CASE WHEN (
        SELECT COUNT(*) 
        FROM заказы 
        WHERE id_клиента = к.id AND дата_заказа >= CURRENT_DATE - INTERVAL '3 months'
    ) > 0 THEN 1 END) AS "Активные за 3 мес",
    COUNT(CASE WHEN (
        SELECT SUM(общая_сумма) 
        FROM заказы 
        WHERE id_клиента = к.id
    ) > 100000 THEN 1 END) AS "Клиенты с высокой выручкой",
    COUNT(CASE WHEN (
        SELECT COUNT(*) 
        FROM заказы 
        WHERE id_клиента = к.id
    ) > 5 THEN 1 END) AS "Частые покупатели",
    -- Средние значения по условию
    AVG(CASE WHEN (
        SELECT SUM(общая_сумма) 
        FROM заказы 
        WHERE id_клиента = к.id
    ) > 50000 THEN (
        SELECT AVG(общая_сумма) 
        FROM заказы 
        WHERE id_клиента = к.id
    ) END) AS "Средний чек высокодоходных",
    -- Общие суммы по условию
    SUM(CASE WHEN (
        SELECT SUM(общая_сумма) 
        FROM заказы 
        WHERE id_клиента = к.id
    ) > 100000 THEN 1 ELSE 0 END * (
        SELECT SUM(общая_сумма) 
        FROM заказы 
        WHERE id_клиента = к.id
    )) AS "Выручка от VIP клиентов"
FROM клиенты к
GROUP BY к.город
HAVING COUNT(*) >= 5  -- только города с 5+ клиентами
ORDER BY "Всего клиентов" DESC;
```

### Паттерн 8: Определение изменений (Change Detection)

#### Проблема
Отслеживание изменений в данных.

#### Решение
Использование LAG/LEAD для сравнения с предыдущими значениями.

```sql
-- Пример: отслеживание изменений в ценах товаров
WITH изменения_цен AS (
    SELECT 
        id_товара,
        цена_на_момент,
        дата_изменения,
        LAG(цена_на_момент) OVER (
            PARTITION BY id_товара 
            ORDER BY дата_изменения
        ) AS предыдущая_цена,
        LAG(дата_изменения) OVER (
            PARTITION BY id_товара 
            ORDER BY дата_изменения
        ) AS дата_предыдущего_изменения
    FROM (
        -- Предположим, у нас есть история цен
        SELECT 
            id_товара,
            цена_на_момент,
            дата_заказа AS дата_изменения  -- упрощение для примера
        FROM позиции_заказа p
        JOIN заказы z ON p.id_заказа = z.id
        WHERE z.дата_заказа >= CURRENT_DATE - INTERVAL '1 year'
        GROUP BY id_товара, цена_на_момент, дата_заказа
    ) AS цены_по_датам
)
SELECT 
    т.название AS товар,
    дата_изменения,
    предыдущая_цена,
    цена_на_момент_цены,
    (цена_на_момент_цены - предыдущая_цена) AS абсолютное_изменение,
    ROUND(((цена_на_момент_цены - предыдущая_цена) * 100.0 / предыдущая_цена), 2) AS процентное_изменение,
    CASE 
        WHEN цена_на_момент_цены > предыдущая_цена THEN 'Рост'
        WHEN цена_на_момент_цены < предыдущая_цена THEN 'Снижение'
        ELSE 'Без изменений'
    END AS тип_изменения,
    EXTRACT(DAY FROM (дата_изменения - дата_предыдущего_изменения)) AS дней_с_последнего_изменения
FROM изменения_цен ic
JOIN товары т ON ic.id_товара = т.id
WHERE предыдущая_цена IS NOT NULL
  AND ABS(процентное_изменение) > 5  -- значительные изменения > 5%
ORDER BY ABS(процентное_изменение) DESC;
```

### Паттерн 9: Построение временных серий

#### Проблема
Анализ данных по временным интервалам.

#### Решение
Использование generate_series и LEFT JOIN для заполнения пропусков.

```sql
-- Пример: создание полного временного ряда без пропусков
WITH 
-- Создание полного ряда дат
все_даты AS (
    SELECT generate_series(
        CURRENT_DATE - INTERVAL '30 days',
        CURRENT_DATE,
        '1 day'::interval
    )::date AS дата
),

-- Данные по дням
продажи_по_дням AS (
    SELECT 
        дата_заказа::date AS дата,
        COUNT(*) AS количество_заказов,
        SUM(общая_сумма) AS выручка
    FROM заказы
    WHERE дата_заказа >= CURRENT_DATE - INTERVAL '30 days'
    GROUP BY дата_заказа::date
)

-- Объединение для заполнения пропусков
SELECT 
    ad.дата,
    COALESCE(pod.количество_заказов, 0) AS количество_заказов,
    COALESCE(pod.выручка, 0) AS выручка,
    -- Скользящее среднее
    AVG(COALESCE(pod.выручка, 0)) OVER (
        ORDER BY ad.дата
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ) AS скользящее_среднее_7_дней
FROM все_даты ad
LEFT JOIN продажи_по_дням pod ON ad.дата = pod.дата
ORDER BY ad.дата;
```

### Паттерн 10: Кумулятивные вычисления

#### Проблема
Вычисление накопительных показателей.

#### Решение
Использование оконных функций с накопительными фреймами.

```sql
-- Пример: накопительная выручка по дням
WITH ежедневные_результаты AS (
    SELECT 
        дата_заказа::date AS дата,
        COUNT(*) AS заказов_в_день,
        SUM(общая_сумма) AS выручка_в_день,
        COUNT(DISTINCT id_клиента) AS уникальных_клиентов
    FROM заказы
    WHERE дата_заказа >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY дата_заказа::date
)
SELECT 
    дата,
    заказов_в_день,
    выручка_в_день,
    уникальных_клиентов,
    -- Кумулятивные значения
    SUM(заказов_в_день) OVER (
        ORDER BY дата
        ROWS UNBOUNDED PRECEDING
    ) AS накопительное_количество_заказов,
    SUM(выручка_в_день) OVER (
        ORDER BY дата
        ROWS UNBOUNDED PRECEDING
    ) AS накопительная_выручка,
    SUM(уникальных_клиентов) OVER (
        ORDER BY дата
        ROWS UNBOUNDED PRECEDING
    ) AS накопительное_количество_клиентов,
    -- Процент от итога
    ROUND(
        SUM(выручка_в_день) OVER (
            ORDER BY дата
            ROWS UNBOUNDED PRECEDING
        ) * 100.0 / 
        SUM(выручка_в_день) OVER (), 
        2
    ) AS процент_от_итога
FROM ежедневные_результаты
ORDER BY дата;
```

## Глава 77: Оптимизация сложных запросов

### Введение

Оптимизация SQL-запросов - критически важный навык для эффективной работы с большими объемами данных. Неправильно оптимизированные запросы могут занимать значительное время, потреблять ресурсы системы и негативно влиять на производительность всей базы данных.

### Методология оптимизации запросов

#### 1. Анализ плана выполнения

```sql
-- Всегда начинайте с EXPLAIN ANALYZE
```

#### 2. Идентификация узких мест

Типичные проблемы:
- Sequential scans на больших таблицах
- Nested loops вместо hash joins
- Отсутствие необходимых индексов
- Неправильное использование функций в условиях
- Избыточное количество данных

### Практические примеры оптимизации

#### Пример 1: Оптимизация JOIN'а

**До оптимизации:**
```sql
-- Медленный запрос: сканирование всей таблицы заказов
SELECT к.фамилия, к.имя, z.дата_заказа, z.общая_сумма
```

```sql
CREATE INDEX idx_клиенты_город ON клиенты(город);
```

#### Пример 2: Оптимизация подзапросов

**До оптимизации:**
```sql
-- Медленный подзапрос, выполняющийся для каждой строки
SELECT фамилия, имя,
       (SELECT COUNT(*) FROM заказы WHERE id_клиента = клиенты.id) AS кол_заказов
FROM клиенты
WHERE (SELECT SUM(общая_сумма) FROM заказы WHERE id_клиента = клиенты.id) > 50000;
```

**После оптимизации:**
```sql
-- Замена подзапросов на JOIN
SELECT к.фамилия, к.имя, COALESCE(стат.кол_заказов, 0) AS кол_заказов
FROM клиенты к
LEFT JOIN (
    SELECT 
        id_клиента,
        COUNT(*) AS кол_заказов,
        SUM(общая_сумма) AS общая_сумма
    FROM заказы
    GROUP BY id_клиента
    HAVING SUM(общая_сумма) > 50000
) AS стат ON к.id = стат.id_клиента
WHERE стат.общая_сумма IS NOT NULL
ORDER BY стат.кол_заказов DESC;
```

#### Пример 3: Оптимизация с GROUP BY

**До оптимизации:**
```sql
-- Неправильное использование группировки
SELECT 
    к.город,
    COUNT(*) AS количество_заказов,
    SUM(з.общая_сумма) AS общая_сумма
FROM клиенты к
JOIN заказы з ON к.id = з.id_клиента
GROUP BY к.город
HAVING SUM(з.общая_сумма) > (
    SELECT AVG(всего) FROM (
        SELECT SUM(общая_сумма) AS всего
        FROM заказы
        GROUP BY id_клиента
    ) AS средние
);
```

**После оптимизации:**
```sql
-- Оптимизированный запрос с предварительной агрегацией
WITH клиентская_статистика AS (
    SELECT 
        к.город,
        SUM(з.общая_сумма) AS общая_сумма_по_городу
    FROM клиенты к
    JOIN заказы з ON к.id = з.id_клиента
    GROUP BY к.id, к.город
),
городская_статистика AS (
    SELECT 
        город,
        SUM(общая_сумма_по_городу) AS общая_сумма,
        COUNT(*) AS количество_клиентов
    FROM клиентская_статистика
    GROUP BY город
),
средние_показатели AS (
    SELECT AVG(сумма_на_клиента) AS средняя_сумма
    FROM (
        SELECT SUM(з.общая_сумма) AS сумма_на_клиента
        FROM заказы з
        GROUP BY з.id_клиента
    ) AS по_клиентам
)
SELECT 
    gs.город,
    gs.количество_клиентов,
    gs.общая_сумма,
    sp.средняя_сумма
FROM городская_статистика gs
CROSS JOIN средние_показатели sp
WHERE gs.общая_сумма > sp.средняя_сумма * 10
ORDER BY gs.общая_сумма DESC;
```

### Оптимизация запросов с оконными функциями

#### Проблема: избыточное использование оконных функций

**До оптимизации:**
```sql
-- Неэффективное использование оконных функций

    к.фамилия,
    к.имя,
    z.дата_заказа,
    z.общая_сумма,
    -- Та же функция выполняется много раз
    RANK() OVER (ORDER BY SUM(z.общая_сумма) OVER (PARTITION BY к.id) DESC) AS ранг_клиента,
    RANK() OVER (ORDER BY AVG(z.общая_сумма) OVER (PARTITION BY к.id) DESC) AS ранг_по_чеку
```

**После оптимизации:**

```sql
    SELECT 
        к.id,
        к.фамилия,
        к.имя,
        SUM(z.общая_сумма) AS общая_сумма,

),

    SELECT *,
           RANK() OVER (ORDER BY общая_сумма DESC) AS ранг_по_сумме,
           RANK() OVER (ORDER BY средний_чек DESC) AS ранг_по_чеку
    FROM клиентская_статистика
)
SELECT 
    фамилия,
    имя,
    общая_сумма,
    средний_чек,
    количество_заказов,
    ранг_по_сумме,
    ранг_по_чеку
FROM ранжированные_клиенты
ORDER BY ранг_по_сумме
LIMIT 100;
```

### Практика: анализ и оптимизация сложного запроса

#### Исходный медленный запрос

```sql
-- Сложный запрос с множеством подзапросов и соединений
SELECT 
    к.фамилия,
    к.имя,
    к.город,
    (SELECT COUNT(*) FROM заказы WHERE id_клиента = к.id AND дата_заказа >= CURRENT_DATE - INTERVAL '1 year') AS заказов_за_год,
    (SELECT SUM(общая_сумма) FROM заказы WHERE id_клиента = к.id AND дата_заказа >= CURRENT_DATE - INTERVAL '1 year') AS сумма_за_год,
    (SELECT AVG(общая_сумма) FROM заказы WHERE id_клиента = к.id AND дата_заказа >= CURRENT_DATE - INTERVAL '1 year') AS средний_чек,
    (SELECT MAX(дата_заказа) FROM заказы WHERE id_клиента = к.id AND дата_заказа >= CURRENT_DATE - INTERVAL '1 year') AS последний_заказ,
    (SELECT COUNT(DISTINCT т.категория) 
     FROM заказы з 
     JOIN позиции_заказа пз ON з.id = пз.id_заказа
     JOIN товары т ON пз.id_товара = т.id
     WHERE з.id_клиента = к.id AND з.дата_заказа >= CURRENT_DATE - INTERVAL '1 year'
    ) AS категорий_покупал
FROM клиенты к
WHERE EXISTS (
    SELECT 1 FROM заказы WHERE id_клиента = к.id AND дата_заказа >= CURRENT_DATE - INTERVAL '1 year'
)
ORDER BY (SELECT SUM(общая_сумма) FROM заказы WHERE id_клиента = к.id AND дата_заказа >= CURRENT_DATE - INTERVAL '1 year') DESC
LIMIT 50;
```

#### Оптимизированный запрос

```sql
-- Оптимизированный вариант с использованием CTE и JOIN
WITH годовая_статистика AS (
    SELECT 
        з.id_клиента,
        COUNT(*) AS заказов_за_год,
        SUM(з.общая_сумма) AS сумма_за_год,
        AVG(з.общая_сумма) AS средний_чек,
        MAX(з.дата_заказа) AS последний_заказ,
        COUNT(DISTINCT т.категория) AS категорий_покупал
    FROM заказы з
    LEFT JOIN позиции_заказа пз ON з.id = пз.id_заказа
    LEFT JOIN товары т ON пз.id_товара = т.id
    WHERE з.дата_заказа >= CURRENT_DATE - INTERVAL '1 year'
    GROUP BY з.id_клиента
)
SELECT 
    к.фамилия,
    к.имя,
    к.город,
    gs.заказов_за_год,
    gs.сумма_за_год,
    ROUND(gs.средний_чек, 2) AS средний_чек,
    gs.последний_заказ,
    gs.категорий_покупал
FROM клиенты к
JOIN годовая_статистика gs ON к.id = gs.id_клиента
ORDER BY gs.сумма_за_год DESC
LIMIT 50;
```

```sql
-- Комплексный анализ поведения пользователей
CREATE VIEW аналитика_поведения_пользователей AS

    SELECT 
        id_пользователя,
        сессия_id,
        MIN(дата_события) AS начало_сессии,

        COUNT(CASE WHEN тип_события = 'click' THEN 1 END) AS кликов,

    FROM события_пользователей
    WHERE дата_события >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY id_пользователя, сессия_id
),
пользовательская_статистика AS (
    SELECT 
        id_пользователя,
        COUNT(*) AS всего_сессий,
        SUM(событий_в_сессии) AS всего_событий,
        SUM(просмотров) AS всего_просмотров,
        SUM(кликов) AS всего_кликов,
        SUM(покупок) AS всего_покупок,
        AVG(событий_в_сессии) AS среднее_событий_на_сессию,
        AVG(просмотров) AS среднее_просмотров_на_сессию,
        AVG(кликов) AS среднее_кликов_на_сессию,
        AVG(продолжительность_сессии_сек) AS средняя_продолжительность_сессии,
        COUNT(CASE WHEN событий_в_сессии = 1 THEN 1 END) AS сессии_с_одним_событием,  -- потенциальные отказы
        COUNT(CASE WHEN покупок > 0 THEN 1 END) AS сессии_с_покупками
    FROM пользовательские_сессии
    GROUP BY id_пользователя
),
ретеншн_анализ AS (
    SELECT 
        id_пользователя,
        -- Определяем дни между регистрацией и последним действием
        EXTRACT(DAYS FROM (CURRENT_DATE - MIN(дата_события))) AS дней_на_платформе,
        -- Количество дней с активностью
        COUNT(DISTINCT дата_события) AS дней_активности,
        -- Коэффициент вовлеченности
        COUNT(DISTINCT дата_события) * 100.0 / EXTRACT(DAYS FROM (CURRENT_DATE - MIN(дата_события) + INTERVAL '1 day')) AS процент_активности
    FROM события_пользователей
    WHERE дата_события >= CURRENT_DATE - INTERVAL '180 days'
    GROUP BY id_пользователя
)
SELECT 
    u.id_пользователя,
    ps.всего_сессий,
    ps.всего_событий,
    ps.всего_просмотров,
    ps.всего_кликов,
    ps.всего_покупок,
    ps.среднее_событий_на_сессию,
    ps.средняя_продолжительность_сессии,
    ra.дней_на_платформе,
    ra.дней_активности,
    ra.процент_активности,
    -- Классификация пользователя
    CASE 
        WHEN ra.процент_активности >= 30 THEN 'Ежедневный пользователь'
        WHEN ra.процент_активности >= 10 THEN 'Частый пользователь'
        WHEN ra.процент_активности >= 3 THEN 'Редкий пользователь'
        ELSE 'Одноразовый пользователь'
    END AS тип_пользователя,
    -- Коэффициент оттока
    ps.сессии_с_одним_событием * 100.0 / NULLIF(ps.всего_сессий, 0) AS коэф_отказа_в_процентах,
    -- Конверсия
    ps.всего_покупок * 100.0 / NULLIF(ps.всего_сессий, 0) AS конверсия_в_сессии_в_процентах
FROM пользовательская_статистика ps
JOIN ретеншн_анализ ra ON ps.id_пользователя = ra.id_пользователя
ORDER BY ps.всего_покупок DESC, ps.всего_событий DESC;
```

### Задача 2: Cohort Analysis (анализ когорт)

```sql
CREATE VIEW когортный_анализ AS
WITH cohorts AS (
    -- Определяем когорту (месяц регистрации клиента)
    SELECT 
        к.id AS id_клиента,
        DATE_TRUNC('month', к.дата_регистрации) AS cohort_date,
        -- Месяцы с момента регистрации
        EXTRACT(YEAR FROM з.дата_заказа) - EXTRACT(YEAR FROM к.дата_регистрации) AS year_diff,
        EXTRACT(MONTH FROM з.дата_заказа) - EXTRACT(MONTH FROM к.дата_регистрации) AS month_diff,
        (EXTRACT(YEAR FROM з.дата_заказа) - EXTRACT(YEAR FROM к.дата_регистрации)) * 12 + 
        (EXTRACT(MONTH FROM з.дата_заказа) - EXTRACT(MONTH FROM к.дата_регистрации)) AS period_number
    FROM клиенты к
    LEFT JOIN заказы з ON к.id = з.id_клиента
    WHERE к.дата_регистрации >= '2022-01-01'
      AND з.дата_заказа >= '2022-01-01'
),
cohort_size AS (
    -- Размер когорты (количество клиентов в каждой когорте)
    SELECT 
        cohort_date,
        COUNT(DISTINCT id_клиента) AS total_users
    FROM (
        SELECT DISTINCT 
            id_клиента,
            DATE_TRUNC('month', дата_регистрации) AS cohort_date
        FROM клиенты
        WHERE дата_регистрации >= '2022-01-01'
    ) AS unique_users
    GROUP BY cohort_date
),
cohort_retention AS (
    -- Выживаемость когорты по периодам
    SELECT 
        c.cohort_date,
        c.period_number,
        COUNT(DISTINCT c.id_клиента) AS active_users
    FROM cohorts c
    GROUP BY c.cohort_date, c.period_number
)
SELECT 
    cr.cohort_date AS когорта,
    cs.total_users AS размер_когорты,
    cr.period_number AS период,
    cr.active_users AS активных_пользователей,
    ROUND(cr.active_users * 100.0 / cs.total_users, 2) AS удержание_в_процентах
FROM cohort_retention cr
JOIN cohort_size cs ON cr.cohort_date = cs.cohort_date
ORDER BY cr.cohort_date, cr.period_number;
```

### Задача 4: Анализ жизненного цикла клиента (Customer Lifecycle)

```sql
CREATE VIEW анализ_жизненного_цикла_клиента AS
WITH клиентский_цикл AS (
    SELECT 
        з.id_клиента,
        к.регистрация_дата,
        MIN(з.дата_заказа) AS первая_покупка,
        MAX(з.дата_заказа) AS последняя_покупка,
        COUNT(*) AS всего_заказов,
        SUM(з.общая_сумма) AS общая_сумма,
        AVG(з.общая_сумма) AS средний_чек,
        -- Длительность взаимодействия
        EXTRACT(DAY FROM (MAX(з.дата_заказа) - MIN(з.дата_заказа))) AS дни_взаимодействия,
        -- Интервал между регистрацией и первой покупкой
        EXTRACT(DAY FROM (MIN(з.дата_заказа) - к.регистрация_дата)) AS дни_до_первой_покупки
    FROM заказы з
    JOIN клиенты к ON з.id_клиента = к.id
    GROUP BY з.id_клиента, к.регистрация_дата
),
стадии_цикла AS (
    SELECT 
        *,
        CASE 
            WHEN дни_до_первой_покупки <= 7 THEN 'Ранние покупатели'
            WHEN дни_до_первой_покупки <= 30 THEN 'Потенциальные покупатели'
            WHEN дни_до_первой_покупки IS NULL THEN 'Не совершили покупку'
            ELSE 'Поздние покупатели'
        END AS стадия_привлечения,
        CASE 
            WHEN последняя_покупка >= CURRENT_DATE - INTERVAL '30 days' THEN 'Активный'
            WHEN последняя_покупка >= CURRENT_DATE - INTERVAL '90 days' THEN 'Снижение активности'
            WHEN последняя_покупка IS NOT NULL THEN 'Неактивный'
            ELSE 'Новый клиент'
        END AS стадия_взаимодействия,
        CASE 
            WHEN всего_заказов = 1 THEN 'Один заказ'
            WHEN всего_заказов <= 3 THEN 'Новые клиенты'
            WHEN всего_заказов <= 10 THEN 'Регулярные клиенты'
            ELSE 'Лояльные клиенты'
        END AS уровень_лояльности,
        -- CLV (Customer Lifetime Value) - упрощенный расчет
        CASE 
            WHEN дни_взаимодействия > 0 THEN общая_сумма * (365.0 / дни_взаимодействия) * 2
            ELSE общая_сумма
        END AS прогноз_CLV
    FROM клиентский_цикл
)
SELECT 
    cl.id_клиента,
    cl.регистрация_дата,
    cl.первая_покупка,
    cl.последняя_покупка,
    cl.всего_заказов,
    cl.общая_сумма,
    cl.средний_чек,
    cl.дни_взаимодействия,
    cl.дни_до_первой_покупки,
    cl.стадия_привлечения,
    cl.стадия_взаимодействия,
    cl.уровень_лояльности,
    ROUND(cl.прогноз_CLV, 2) AS прогноз_CLV,
    -- Рекомендуемые действия
    CASE 
        WHEN cl.стадия_взаимодействия = 'Снижение активности' AND cl.уровень_лояльности = 'Лояльные клиенты' THEN 'Ретеншн-кампания'
        WHEN cl.стадия_привлечения = 'Не совершили покупку' THEN 'Реклама скидок для новичков'
        WHEN cl.стадия_взаимодействия = 'Неактивный' AND cl.уровень_лояльности = 'Лояльные клиенты' THEN 'Кампания возврата'
        WHEN cl.стадия_привлечения = 'Потенциальные покупатели' THEN 'Таргетированная реклама'
        ELSE 'Обычное обслуживание'
    END AS рекомендуемое_действие
FROM стадии_цикла cl
ORDER BY cl.прогноз_CLV DESC;
```

```sql
CREATE VIEW анализ_ассоциаций_товаров AS
WITH товарные_пары AS (
    -- Найти пары товаров, которые часто заказываются вместе
    SELECT 
        пз1.id_товара AS товар1_id,
        пз2.id_товара AS товар2_id,
        COUNT(*) AS совместных_заказов,
        COUNT(DISTINCT пз1.id_заказа) AS заказов_товара1,
        COUNT(DISTINCT пз2.id_заказа) AS заказов_товара2
    FROM позиции_заказа пз1
    JOIN позиции_заказа пз2 ON пз1.id_заказа = пз2.id_заказа
    WHERE пз1.id_товара < пз2.id_товара  -- избегаем дубликатов (A,B) и (B,A)
    GROUP BY пз1.id_товара, пз2.id_товара
    HAVING COUNT(*) > 1  -- только часто заказываемые вместе
)
SELECT 
    t1.название AS товар1,
    t2.название AS товар2,
    tp.совместных_заказов,
    tp.заказов_товара1 AS заказов_товара1_всего,
    tp.заказов_товара2 AS заказов_товара2_всего,
    -- Поддержка (support) - насколько часто товары покупаются вместе
    ROUND(tp.совместных_заказов * 100.0 / (
        SELECT COUNT(DISTINCT id_заказа) FROM позиции_заказа
    ), 2) AS поддержка_в_процентах,
    -- Уверенность (confidence) - вероятность покупки товара2 при покупке товара1
    ROUND(tp.совместных_заказов * 100.0 / tp.заказов_товара1, 2) AS уверенность_в_процентах,
    -- Подъем (lift) - насколько чаще товары покупаются вместе, чем отдельно
    ROUND(
        (tp.совместных_заказов * (
            SELECT COUNT(DISTINCT id_заказа) FROM позиции_заказа
        )) / (tp.заказов_товара1 * tp.заказов_товара2), 2
    ) AS лифт,
    CASE 
        WHEN лифт > 2 THEN 'Сильная связь'
        WHEN лифт > 1.5 THEN 'Средняя связь'
        WHEN лифт > 1.2 THEN 'Слабая связь'
        ELSE 'Случайная связь'
    END AS сила_связи
FROM товарные_пары tp
JOIN товары t1 ON tp.товар1_id = t1.id
JOIN товары t2 ON tp.товар2_id = t2.id
WHERE tp.совместных_заказов >= 3  -- минимум 3 совместных заказа
ORDER BY tp.совместных_заказов DESC, лифт DESC
LIMIT 50;
```

```sql
CREATE VIEW прогнозирование_поведения_клиентов AS
WITH клиентская_статистика AS (
    SELECT 
        з.id_клиента,
        COUNT(*) AS всего_заказов,
        SUM(з.общая_сумма) AS общая_сумма,
        AVG(з.общая_сумма) AS средний_чек,
        MAX(з.дата_заказа) AS последняя_покупка,
        MIN(з.дата_заказа) AS первая_покупка,
        -- Частота покупок (в днях)
        CASE 
            WHEN COUNT(*) > 1 THEN 
                EXTRACT(DAY FROM (MAX(з.дата_заказа) - MIN(з.дата_заказа))) / (COUNT(*) - 1)
            ELSE NULL
        END AS средняя_частота_в_днях,
        -- Сколько дней прошло с последней покупки
        EXTRACT(DAY FROM (CURRENT_DATE - MAX(з.дата_заказа))) AS дней_без_покупки
    FROM заказы з
    GROUP BY з.id_клиента
),
clv_метрики AS (
    SELECT 
        *,
        CASE 
            WHEN средняя_частота_в_днях IS NOT NULL AND средняя_частота_в_днях > 0 THEN
                (общая_сумма / (EXTRACT(DAY FROM (последняя_покупка - первая_покупка)) + 1)) * 365 * 2
            ELSE общая_сумма
        END AS CLV_прогноз,
        -- Вероятность оттока (упрощенная модель)
        CASE 
            WHEN дней_без_покупки > COALESCE(средняя_частота_в_днях, 30) * 2 THEN 0.8
            WHEN дней_без_покупки > COALESCE(средняя_частота_в_днях, 30) THEN 0.5
            ELSE 0.1
        END AS вероятность_оттока,
        -- Потенциал роста (отношение общей суммы к числу заказов)
        общая_сумма / NULLIF(всего_заказов, 0) AS средняя_ценность_заказа
    FROM клиентская_статистика
)
SELECT 
    к.фамилия,
    к.имя,
    к.город,
    cm.всего_заказов,
    cm.общая_сумма,
    cm.средний_чек,
    cm.средняя_частота_в_днях,
    cm.дней_без_покупки,
    ROUND(cm.CLV_прогноз, 2) AS прогноз_CLV,
    ROUND(cm.вероятность_оттока * 100, 2) AS вероятность_оттока_в_процентах,
    ROUND(cm.средняя_ценность_заказа, 2) AS средняя_ценность_заказа,
    -- Рекомендация по удержанию
    CASE 
        WHEN cm.вероятность_оттока > 0.6 THEN 'Срочная кампания удержания'
        WHEN cm.вероятность_оттока > 0.3 THEN 'Периодические контакты'
        WHEN cm.общая_сумма > 100000 THEN 'VIP-обслуживание'
        ELSE 'Стандартное обслуживание'
    END AS рекомендуемая_стратегия
FROM clv_метрики cm
JOIN клиенты к ON cm.id_клиента = к.id
ORDER BY cm.CLV_прогноз DESC
LIMIT 100;
```

### Задача 7: Сравнительный анализ и динамика

```sql
-- Сравнение показателей по периодам
CREATE VIEW динамика_бизнес_показателей AS
WITH ежемесячная_статистика AS (
    SELECT 
        EXTRACT(YEAR FROM дата_заказа) AS год,
        EXTRACT(MONTH FROM дата_заказа) AS месяц,
        COUNT(*) AS количество_заказов,
        SUM(общая_сумма) AS выручка,
        AVG(общая_сумма) AS средний_чек,
        COUNT(DISTINCT id_клиента) AS уникальных_клиентов,
        SUM(количество_товаров) AS товарооборот
    FROM (
        SELECT 
            дата_заказа,
            общая_сумма,
            id_клиента,
            (SELECT SUM(количество) FROM позиции_заказа WHERE id_заказа = заказы.id) AS количество_товаров
        FROM заказы
    ) AS заказы_с_товарами
    GROUP BY EXTRACT(YEAR FROM дата_заказа), EXTRACT(MONTH FROM дата_заказа)
),
динамика AS (
    SELECT 
        *,
        LAG(количество_заказов) OVER (ORDER BY год, месяц) AS пред_заказы,
        LAG(выручка) OVER (ORDER BY год, месяц) AS пред_выручка,
        LAG(средний_чек) OVER (ORDER BY год, месяц) AS пред_средний_чек,
        LAG(уникальных_клиентов) OVER (ORDER BY год, месяц) AS пред_клиенты,
        -- Рост в процентах
        ROUND(((количество_заказов - LAG(количество_заказов) OVER (ORDER BY год, месяц)) * 100.0 / 
               NULLIF(LAG(количество_заказов) OVER (ORDER BY год, месяц), 0)), 2) AS рост_заказов_в_процентах,
        ROUND(((выручка - LAG(выручка) OVER (ORDER BY год, месяц)) * 100.0 / 
               NULLIF(LAG(выручка) OVER (ORDER BY год, месяц), 0)), 2) AS рост_выручки_в_процентах
    FROM ежемесячная_статистика
)
SELECT 
    год,
    месяц,
    количество_заказов,
    выручка,
    средний_чек,
    уникальных_клиентов,
    товарооборот,
    рост_заказов_в_процентах,
    рост_выручки_в_процентах,
    CASE 
        WHEN рост_выручки_в_процентах > 10 THEN 'Сильный рост'
        WHEN рост_выручки_в_процентах > 5 THEN 'Умеренный рост'
        WHEN рост_выручки_в_процентах > 0 THEN 'Медленный рост'
        WHEN рост_выручки_в_процентах > -5 THEN 'Небольшое падение'
        ELSE 'Сильное падение'
    END AS оценка_динамики,
    -- Средние показатели за последние 3 месяца
    AVG(количество_заказов) OVER (
        ORDER BY год, месяц
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS среднее_за_3_месяца_заказов,
    AVG(выручка) OVER (
        ORDER BY год, месяц
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS среднее_за_3_месяца_выручки
FROM динамика
ORDER BY год DESC, месяц DESC;
```

### 2. Пакетная обработка данных

#### Пакетное обновление с курсорами

```sql
-- Пример пакетного обновления для больших таблиц
DO $$
DECLARE
    batch_size INTEGER := 1000;
    processed_count INTEGER := 0;
    total_count INTEGER;
BEGIN
    -- Получаем общее количество записей для обработки
    SELECT COUNT(*) INTO total_count
    FROM клиенты
    WHERE дата_регистрации < CURRENT_DATE - INTERVAL '1 year'
      AND статус IS DISTINCT FROM 'архивный';
    
    -- Пакетная обработка
    WHILE processed_count < total_count LOOP
        UPDATE клиенты
        SET статус = 'архивный'
        WHERE id IN (
            SELECT id
            FROM клиенты
            WHERE дата_регистрации < CURRENT_DATE - INTERVAL '1 year'
              AND статус IS DISTINCT FROM 'архивный'
            LIMIT batch_size
            OFFSET processed_count
        );
        
        GET DIAGNOSTICS processed_count = ROW_COUNT;
        
        RAISE NOTICE 'Обработано % из % записей', processed_count, total_count;
        
        -- Даем системе время на другие операции
        PERFORM pg_sleep(0.1);
    END LOOP;
    
    RAISE NOTICE 'Пакетное обновление завершено';
END $$;
```

```sql
    )
    SELECT 'Количество заказов'::VARCHAR, заказов::NUMERIC, 'Период'::VARCHAR
    FROM статистика
    UNION ALL
    SELECT 'Общая выручка'::VARCHAR, выручка::NUMERIC, 'Период'::VARCHAR
    FROM статистика
    UNION ALL
    SELECT 'Средний чек'::VARCHAR, средний_чек::NUMERIC, 'Период'::VARCHAR

    UNION ALL
    SELECT 'Уникальные клиенты'::VARCHAR, уникальных_клиентов::NUMERIC, 'Период'::VARCHAR
    FROM статистика;
END;
```

```sql
    SELECT 
        id_клиента,
        COUNT(*) AS количество_заказов,
        SUM(общая_сумма) AS общая_сумма,

    GROUP BY id_клиента

    SELECT 
        *,
        CASE 
            WHEN общая_сумма > 100000 THEN 'VIP'
            WHEN количество_заказов > 10 THEN 'Частый'
            WHEN последний_заказ >= CURRENT_DATE - INTERVAL '30 days' THEN 'Активный'
            ELSE 'Обычный'
        END AS сегмент
    FROM пользовательская_статистика
)
SELECT 
    id_клиента,
    сегмент,
    количество_заказов,
    общая_сумма

    AVG(a.общая_сумма) AS средняя_сумма
```

```sql
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO очередь_изменений (тип_события, таблица, id_записи, новые_данные)
        VALUES ('insert', TG_TABLE_NAME, NEW.id, to_jsonb(NEW));
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO очередь_изменений (тип_события, таблица, id_записи, старые_данные, новые_данные)
        VALUES ('update', TG_TABLE_NAME, NEW.id, to_jsonb(OLD), to_jsonb(NEW));
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO очередь_изменений (тип_события, таблица, id_записи, старые_данные)
        VALUES ('delete', TG_TABLE_NAME, OLD.id, to_jsonb(OLD));
        RETURN OLD;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- Создание триггера для таблицы клиентов
CREATE TRIGGER tr_клиенты_очереди
    AFTER INSERT OR UPDATE OR DELETE ON клиенты
    FOR EACH ROW
    EXECUTE FUNCTION добавить_очередь();
```

#### Обработка очереди из SQL

```sql
-- Функция для обработки сообщений
CREATE OR REPLACE FUNCTION обработать_очередь_изменений(
    p_limit INTEGER DEFAULT 100
)
RETURNS INTEGER AS $$
DECLARE
    processed_count INTEGER := 0;
    record_id INTEGER;
    event_type VARCHAR;
    event_data JSONB;
BEGIN
    -- Цикл по сообщениям в очереди
    FOR record_id, event_type, event_data IN
        SELECT id, тип_события, новые_данные
        FROM очередь_изменений
        WHERE статус = 'pending'
        ORDER BY дата_создания
        LIMIT p_limit
    LOOP
        -- Здесь будет логика обработки сообщения
        -- Например, отправка в Kafka, обновление внешней системы и т.д.
        
        -- В примере просто обновляем статус
        UPDATE очередь_изменений
        SET статус = 'done', обработан = CURRENT_TIMESTAMP
        WHERE id = record_id;
        
        processed_count := processed_count + 1;
    END LOOP;
    
    RETURN processed_count;
END;
$$ LANGUAGE plpgsql;
```

```sql
-- Комплексное представление для аналитики продаж
CREATE VIEW аналитика_продаж_bi AS
SELECT 
    DATE_TRUNC('day', з.дата_заказа) AS дата,
    DATE_TRUNC('month', з.дата_заказа) AS месяц,
    DATE_TRUNC('quarter', з.дата_заказа) AS квартал,
    DATE_TRUNC('year', з.дата_заказа) AS год,
    к.город AS регион,
    т.категория AS категория_товара,
    COUNT(*) AS количество_заказов,
    SUM(з.общая_сумма) AS выручка,
    AVG(з.общая_сумма) AS средний_чек,
    COUNT(DISTINCT з.id_клиента) AS уникальных_клиентов,
    SUM(з.общая_сумма) / COUNT(DISTINCT з.id_клиента) AS arpc,
    SUM(количество_товаров) AS количество_товаров_в_продажах
FROM заказы з
JOIN клиенты к ON з.id_клиента = к.id
JOIN (
    SELECT 
        id_заказа,
        SUM(количество) AS количество_товаров
    FROM позиции_заказа
    GROUP BY id_заказа
) AS позиции_сумма ON з.id = позиции_сумма.id_заказа
JOIN позиции_заказа пз ON з.id = пз.id_заказа
JOIN товары т ON пз.id_товара = т.id
WHERE з.дата_заказа >= CURRENT_DATE - INTERVAL '2 years'
GROUP BY 
    DATE_TRUNC('day', з.дата_заказа),
    DATE_TRUNC('month', з.дата_заказа),
    DATE_TRUNC('quarter', з.дата_заказа),
    DATE_TRUNC('year', з.дата_заказа),
    к.город,
    т.категория;

-- Индексы для оптимизации запросов BI-инструментов
CREATE INDEX idx_аналитика_продаж_дата_регион ON аналитика_продаж_bi(дата, регион);
CREATE INDEX idx_аналитика_продаж_год_категория ON аналитика_продаж_bi(год, категория_товара);
```

```sql
CREATE OR REPLACE FUNCTION отправить_сообщение(
    p_тема VARCHAR,
    p_тип VARCHAR,
    p_данные JSONB,
    p_получатель VARCHAR DEFAULT NULL
)

DECLARE
    new_id INTEGER;

    p_тема VARCHAR DEFAULT NULL,
    p_получатель VARCHAR DEFAULT NULL,
    p_limit INTEGER DEFAULT 10
)
RETURNS SETOF сообщения_обмена AS $$
BEGIN
    RETURN QUERY
    UPDATE сообщения_обмена
    SET 
        статус = 'processing',
        попыток_обработки = попыток_обработки + 1,
        дата_обработки = CURRENT_TIMESTAMP
    WHERE (статус = 'new' OR (статус = 'failed' AND попыток_обработки < максимальных_попыток))
      AND (p_тема IS NULL OR тема = p_тема)
      AND (p_получатель IS NULL OR получатель = p_получатель)
    RETURNING *;
END;
$$ LANGUAGE plpgsql;
```

```sql
    p_id INTEGER,
    p_новые_метаданные JSONB DEFAULT NULL,
    p_новые_теги JSONB DEFAULT NULL
)
RETURNS BOOLEAN AS $$
DECLARE
    rows_affected INTEGER;
BEGIN
    UPDATE файлы
    SET 
        метаданные = COALESCE(p_новые_метаданные, метаданные),
        теги = COALESCE(p_новые_теги, теги)
    WHERE id = p_id AND статус = 'active';
    
    GET DIAGNOSTICS rows_affected = ROW_COUNT;
    
    RETURN rows_affected > 0;
END;
$$ LANGUAGE plpgsql;
```
